==========================================
Response Structure
==========================================
/api                    (get)
    message : API is running

/api/login              (post)
    message : Login successful
    user :
        id : id
        name : name + lname
        email : db_email
        permission : perms

==========================================

/api/position           (get)
    List[Position]
        id: str
        name: str
        permission: str

/api/position           (post)
    msg: str
    info: str

/api/position/{id}      (put)
    msg: str
    info: str

/api/position/{id}      (delete)
    msg: str
    info: str

==========================================

/api/user               (get)
    List[Userdata]
        id: str
        f_name: str
        l_name: str
        email: str
        passwd: str
        pos: str        (Position ID)
        dep: str        (Department ID)
        isemp: bool
        salary: float

/api/user-linked        (get)
    List[Userdata]
        id: str
        f_name: str
        l_name: str
        email: str
        passwd: str
        pos: str        (Position name)
        dep: str        (Department name)
        isemp: bool
        salary: float
