SELECT
    VT."NAME" AS "VEHICLE_TYPE",
    V."PLATE_NO",
    V."SEAT",
    COUNT(TL."LOG_ID") AS "TRIPS"
FROM 
    "VEHICLE_TYPE" VT
    JOIN "VEHICLE" V
        ON V."TYPE" = VT."ID"
    LEFT JOIN "TRIP_LOG" TL
        ON TL."VEHICLE" = V."ID"
            AND TL."DATE" >= DATE '2025-04-01'
            AND TL."DATE" < DATE '2025-05-01'
GROUP BY
    VT."ID",
    VT."NAME",
    V."ID",
    V."SEAT",
    V."PLATE_NO"
ORDER BY
    VT."ID",
    V."ID";
