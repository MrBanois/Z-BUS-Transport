# Download Things (Oracle Database Free)
docker pull gvenzl/oracle-free:slim

# Create persistent volume
docker volume create OracleDBData

# Run The Thing
docker run -d --name oracle-db -p 1521:1521 -e ORACLE_PASSWORD="Admin" -v OracleDBData:/opt/oracle/oradata gvenzl/oracle-free:slim

# Read Logs (Optional)
docker logs -f oracle-db

# Open SQL+ inside the container
docker exec -it oracle-db sqlplus system/Admin@FREEPDB1

#Inside SQL+

GRANT CREATE SESSION TO DEV;
GRANT CREATE TABLE TO DEV;
GRANT CREATE VIEW TO DEV;
GRANT CREATE SEQUENCE TO DEV;
GRANT CREATE PROCEDURE TO DEV;
GRANT CREATE TRIGGER TO DEV;
GRANT CREATE TYPE TO DEV;
GRANT CREATE SYNONYM TO DEV;
GRANT CREATE MATERIALIZED VIEW TO DEV;

CREATE USER Nope IDENTIFIED BY "Koi";
GRANT DEV TO Nope;
ALTER USER Nope QUOTA UNLIMITED ON SYSTEM;
ALTER USER Nope QUOTA UNLIMITED ON USERS;
