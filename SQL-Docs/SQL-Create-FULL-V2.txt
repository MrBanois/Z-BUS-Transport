--================================================
--	Master Files
--================================================

DROP TABLE "VEHICLE_TYPE" CASCADE CONSTRAINTS PURGE;
DROP TABLE "POSITION" CASCADE CONSTRAINTS PURGE;
DROP TABLE "STATUS" CASCADE CONSTRAINTS PURGE;
DROP TABLE "DEPARTMENT" CASCADE CONSTRAINTS PURGE;
DROP TABLE "STATION" CASCADE CONSTRAINTS PURGE;
DROP TABLE "VEHICLE" CASCADE CONSTRAINTS PURGE;
DROP TABLE "USER" CASCADE CONSTRAINTS PURGE;
DROP TABLE "ROUTE" CASCADE CONSTRAINTS PURGE;
DROP TABLE "ROUTE_DETAIL" CASCADE CONSTRAINTS PURGE;
DROP TABLE "SCHEDULE" CASCADE CONSTRAINTS PURGE;

CREATE TABLE "VEHICLE_TYPE" (
    "ID" CHAR(5) PRIMARY KEY,
    "NAME" VARCHAR2(20)
);

CREATE TABLE "POSITION" (
    "ID" CHAR(5) PRIMARY KEY,
    "NAME" VARCHAR2(20),
    "PERMISSION" VARCHAR2(10)
);

CREATE TABLE "STATUS" (
    "ID" CHAR(1) PRIMARY KEY,
    "NAME" VARCHAR2(15)
);

CREATE TABLE "DEPARTMENT" (
    "ID" CHAR(5) PRIMARY KEY,
    "NAME" VARCHAR2(20)
);

CREATE TABLE "STATION" (
    "ID" CHAR(5) PRIMARY KEY,
    "NAME" VARCHAR2(20),
    "IS_ACTIVE" CHAR(1), --BOOLEAN
    CHECK ("IS_ACTIVE" IN ('T','F')) --REMOVE IF BOOL DTYPE EXISTS
);

CREATE TABLE "VEHICLE" (
    "ID" CHAR(5) PRIMARY KEY,
    "PLATE_NO" VARCHAR2(10),
    "SEAT" NUMBER(2),
    "TYPE" CHAR(5),
    "IS_ACTIVE" CHAR(1) , --BOOLEAN
    CHECK ("IS_ACTIVE" IN ('T','F')), --REMOVE IF BOOL DTYPE EXISTS
    CONSTRAINT FK_VEH_TYPE FOREIGN KEY ("TYPE") REFERENCES "VEHICLE_TYPE"("ID")
);

CREATE TABLE "USER" (
    "ID" CHAR(10) PRIMARY KEY,
    "F_NAME" VARCHAR2(20),
    "L_NAME" VARCHAR2(20),
    "EMAIL" VARCHAR2(20),
    "PASSWORD" CHAR(32), --MD5 HASH 
    "POS" CHAR(5), --FK POSITION.ID
    "DEP" CHAR(5), -- FK DEPARTMENT.ID
    CONSTRAINT FK_USER_POS FOREIGN KEY ("POS") REFERENCES "POSITION"("ID"),
    CONSTRAINT FK_USER_DEP FOREIGN KEY ("DEP") REFERENCES "DEPARTMENT"("ID")
);

CREATE TABLE "ROUTE" (
    "ID" CHAR(5) PRIMARY KEY,
    "NAME" VARCHAR2(30),
    "IS_ACTIVE" CHAR(1), --BOOLEAN
    CHECK ("IS_ACTIVE" IN ('T','F')) --REMOVE IS BOOL DTYPE EXISTS
);

CREATE TABLE "ROUTE_DETAIL" (
    "STATION_NO" CHAR(2), --PK
    "RID" CHAR(5), --FK+PK ROUTE.ID
    "STATION" CHAR(5), --FK STATION.ID
    "TIME" TIMESTAMP,
    CONSTRAINT FK_ROUTE_DETAIL_RID FOREIGN KEY ("RID") REFERENCES "ROUTE"("ID"),
    CONSTRAINT FK_ROUTE_DETAIL_SID FOREIGN KEY ("STATION") REFERENCES "STATION"("ID"),
    CONSTRAINT PK_ROUTE_DETAIL_SNO_RID PRIMARY KEY ("STATION_NO","RID")
);

CREATE TABLE "SCHEDULE" (
    "RID" CHAR(5), --FK+PK ROUTE.ID
    "TRIP_NO" CHAR(2), --PK
    "DRIVER" CHAR(10), --FK USER.ID
    "VEHICLE" CHAR(5), --FK VEHICLE.ID
    "TIME_START" TIMESTAMP,
    "IS_ACTIVE" CHAR(1), --BOOLEAN
    CHECK ("IS_ACTIVE" IN ('T','F')), --REMOVE IF BOOL DTYPE EXISTS
    CONSTRAINT FK_SCHEDULE_RID FOREIGN KEY ("RID") REFERENCES "ROUTE"("ID"),
    CONSTRAINT FK_SCHEDULE_DRI FOREIGN KEY ("DRIVER") REFERENCES "USER"("ID"),
    CONSTRAINT FK_SCHEDULE_VEH FOREIGN KEY ("VEHICLE") REFERENCES "VEHICLE"("ID"),
    CONSTRAINT PK_SCHEDULE_RID_TRIP PRIMARY KEY ("RID","TRIP_NO")
);






--================================================
--	Transactions
--================================================

DROP TABLE "BOOKING" CASCADE CONSTRAINTS PURGE;
DROP TABLE "BOOKING_DETAIL" CASCADE CONSTRAINTS PURGE;
DROP TABLE "TRIP_LOG" CASCADE CONSTRAINTS PURGE;
DROP TABLE "TRIP_LOG_DETAIL" CASCADE CONSTRAINTS PURGE;

CREATE TABLE "BOOKING" (
    "BOOKING_ID" CHAR(13) PRIMARY KEY,
    "USER" CHAR(10), --FK USER.ID
    "DATE" TIMESTAMP,
    CONSTRAINT FK_BOOKING_USR FOREIGN KEY ("USER") REFERENCES "USER"("ID")
);

CREATE TABLE "BOOKING_DETAIL" (
    "BID" CHAR(13), --PK FK BOOKING.BOOKING_ID
    "NO" CHAR(2), --PK
    "ROUTE" CHAR(5), --FK SCHEDULE.RID
    "SCHEDULE" CHAR(2), --FK SCHEDULE.TRIP_NO
    "STARTNO" CHAR(2), --FK ROUTE_DETAIL.STATION_NO
    "SRID" CHAR(5), --FK ROUTE_DETAIL.RID
    "ENDNO" CHAR(2), --FK ROUTE_DETAIL.STATION_NO
    "ERID" CHAR(5), --FK ROUTE_DETAIL.RID
    "SEAT" NUMBER(2),
    "QR_STRING" VARCHAR2(50),
    "CURRENT_STATUS" CHAR(1),
    CONSTRAINT FL_BOOKING_DETAIL_STATUS FOREIGN KEY ("CURRENT_STATUS") REFERENCES "STATUS"("ID"),
    CONSTRAINT FK_BOOKING_DETAIL_BID FOREIGN KEY ("BID") REFERENCES "BOOKING"("BOOKING_ID"),
    CONSTRAINT FK_BOOKING_DETAIL_ROUTE FOREIGN KEY ("ROUTE","SCHEDULE") REFERENCES "SCHEDULE"("RID","TRIP_NO"),
    CONSTRAINT FK_BOOKING_DETAIL_STARTNO FOREIGN KEY ("STARTNO","SRID") REFERENCES "ROUTE_DETAIL"("STATION_NO","RID"),
    CONSTRAINT FK_BOOKING_DETAIL_ENDNO FOREIGN KEY ("ENDNO","ERID") REFERENCES "ROUTE_DETAIL"("STATION_NO","RID"),
    CONSTRAINT PK_BOOKING_DETAIL_BID_NO PRIMARY KEY("BID","NO")
);

CREATE TABLE "TRIP_LOG" (
    "LOG_ID" CHAR(10) PRIMARY KEY,
    "DRIVER" CHAR(10), --FK USER.ID
    "VEHICLE" CHAR(5), --FK VEHICLE.ID
    "DATE" DATE,
    "TIME_START" TIMESTAMP,
    "TIME_END" TIMESTAMP,
    CONSTRAINT FK_TRIP_LOG_DRIVER FOREIGN KEY ("DRIVER") REFERENCES "USER"("ID"),
    CONSTRAINT FK_TRIP_LOG_VEHICLE FOREIGN KEY ("VEHICLE") REFERENCES "VEHICLE"("ID")
);

CREATE TABLE "TRIP_LOG_DETAIL"(
    "LID" CHAR(10), --PK FK TRIP_LOG.LOG_ID
    "BID" CHAR(13), --PK FK BOOKING_DETAIL.BID
    "NO" CHAR(2), --PK FK BOOKING_DETAIL.NO
    "TIME_GETIN" TIMESTAMP,
    "TIME_GETOUT" TIMESTAMP,
    "PASSENGER" NUMBER(3),
    CONSTRAINT FK_TRIP_LOG_DETAIL_LID FOREIGN KEY ("LID") REFERENCES "TRIP_LOG"("LOG_ID"),
    CONSTRAINT FK_TRIP_LOG_DETAIL_BID_NO FOREIGN KEY ("BID","NO") REFERENCES "BOOKING_DETAIL"("BID","NO"),
    CONSTRAINT PK_TRIP_LOG_DETAIL_LID_BID_NO PRIMARY KEY ("LID","BID","NO")
);
